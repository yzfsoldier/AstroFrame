# Settings for AstroQuest on the Steam Frame (astroquest.sh), one key=value a line. Lines
# starting with # are ignored; what is commented out shows the value that applies when the line
# is left out. The same keys can be given as environment variables in the game's Steam launch
# options, as ASTROQUEST_<KEY>=value %command% (ASTROQUEST_FOV=90 %command%), which win over
# this file.

# The display refresh rate to ask the headset for (72, 90, 120, 144), where its runtime lets a
# program choose; otherwise the rate set in SteamVR applies. It decides which frame rates the
# game can keep in step with the display: 45 or 30 frames a second at 90 Hz, 60, 40 or 30 at
# 120, 36 at 72. 0: leave the rate alone.
#refresh_rate=90

# How many of the display's refreshes every frame of the game is given (2 is the game's own
# way). Left out: the emulator chooses, as on the Quest.
#pace=

# The most frames a second (60 is the console's own). Left out: as many as the pace makes.
#fps=

# The size the game draws its scene at, per eye: 816 (x870), 960 (x1080), 1200 (x1280),
# 1440 (x1536), larger widths (1800, 2160 ... as on the PC, with the GPU time and memory that
# takes), or game for the game's own choice. Left out: the emulator chooses, as large as the GPU
# manages at the pace.
#resolution=

# How much of the headset's field of view the game draws, in percent, 50 to 100: less puts the
# same pixels over fewer degrees, a sharper picture with a soft border where the rest was.
#fov=85

# Sharpening of every frame of the game, 0 (none) to 1.
#sharpen=0.6

# Foveated rendering: the scene is shaded more coarsely away from where you look, where the eye
# sees less sharply anyway, so the GPU has time for a larger picture. 0: off; 1: mild (every
# pixel out to half way to the edge, 2x2 beyond); 2: medium (2x2 from 45%, 4x4 from 75%);
# 3 (the default): strong (2x2 from 35%, 4x4 from 60%).
#foveation=3

# 1: with foveation on, the fine middle follows where the eyes look (the Frame's eye tracking,
# where SteamVR passes it on); 0: it stays in the middle of the picture.
#eye_tracking=1

# How steady the eye tracking is kept: the readings tremble, and are smoothed while the gaze
# rests (a jump of the eyes is followed at once). The number is how fast a resting gaze may
# drift, in Hz: smaller is steadier and slower, larger follows more closely. 0: as the headset
# says.
#gaze_smoothing=1

# Degrees to move where the headset says you look by, right and up (negative: left, down),
# for when the sharp middle sits a little beside where you look (foveation_colors=1 shows it):
# gaze_offset=2,-1
#gaze_offset=0,0

# 1: a debug overlay in the headset: frame rate, what frames take, the display's rate and the
# refreshes a frame is given, the size the scene is drawn at, how busy the GPU is, draws a
# frame, multisampling, foveation and where the eyes look (updated twice a second).
#overlay=0

# Where the overlay's text sits in each eye's picture: across and down, 0 to 1 each (0.5,0.5 is
# the middle). The further out, the more the lenses blur it.
#overlay_position=0.5,0.38

# 1: with foveation on, its rings in colour over the picture: green where every pixel is
# shaded, yellow where one colour does for 2x2 pixels, red for 4x4; a small white ring where
# the eyes look.
#foveation_colors=0

# The most samples a pixel gets (1, 2, 4, 8; the game draws with 8). 1 (the default): no
# multisampling, the fastest; the emulator smooths edges itself, unless antialias=0. 4: the
# game's own multisampling, sharper edges, more GPU time.
#msaa=1
#antialias=1

# 1: with msaa=1, the GPU driver draws without its tiled mode, which was faster for this game on
# the Quest's Adreno GPU without multisampling. 0: the driver chooses.
#tu_sysmem=1

# 1: barriers added between the writes to and reads from shared memory in the game's compute
# shaders (needed on GPUs that run fewer than 64 threads in step; not the Frame's, so off).
#lds_barriers=0

# 0: the game counts time in frames, as on the console (slow motion whenever a frame takes
# longer than a sixtieth of a second).
#real_time=1

# 0: do not use hand tracking to place a gamepad in the game.
#hands=1

# 0: the headset's own controllers never stand in for a gamepad.
#controllers=1

# Which of the headset's controllers is the controller in the game: right or left.
#controller_hand=right

# How many degrees the view turns for each flick of the right stick with the left grip (L1 on a
# gamepad) held. 0: never.
#turn=30

# How far beyond the next refresh the head pose given to the game is predicted, in
# milliseconds (0 to 80).
#predict_ms=20

# 0: the right stick no longer doubles as a finger on the touchpad.
#stick_touchpad=1

# Makes what the microphone hears that many times louder for the game (0.1 to 30).
#mic_gain=1

# 0: the game's surround sound is folded down to left and right.
#surround=1

# 0: the game is never made to wait while the headset is off the head.
#pause=1

# The language the game is played in, as a language tag (fr-FR, de-DE, pt-BR, ja-JP ...).
# Left out: the system's (LANG). The game has 28 languages and is in English for any other.
#language=

# 1: also open a window with the game on the desktop (for testing on a Linux PC with a headset;
# the headset itself shows only what is in it).
#window=0

# Seconds to wait for a headset before the game starts without one. 0: do not wait.
#wait=60

# Where the game is, if it is not in one of the games folders (see README.md): its
# eboot.bin or its folder.
#game=/home/steamuser/AstroQuest/games/CUSA12392/eboot.bin

# Where saves, logs and the emulator's settings are kept. Left out: ~/.local/share/AstroQuest,
# which a new upload of the program with the Devkit Client leaves alone.
#data=
